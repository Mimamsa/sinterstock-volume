cmake_minimum_required(VERSION 3.16)
project(pointcloud_compare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets Gui Concurrent)
find_package(Qhull CONFIG REQUIRED)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Boost REQUIRED)
add_library(mesh_volume STATIC src/mesh_volume.cpp)
target_link_libraries(mesh_volume PUBLIC Qt5::Core Eigen3::Eigen Qhull::qhullcpp Boost::boost)
target_compile_options(mesh_volume PRIVATE -Wall -Wextra -Wpedantic)
add_executable(calibrate_clouds src/calibrate.cpp)
target_link_libraries(calibrate_clouds PRIVATE Qt5::Core Eigen3::Eigen)
target_compile_options(calibrate_clouds PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pointcloud_compare src/main.cpp src/cloud.cpp src/surface_filter.cpp src/columns.cpp src/ply_export.cpp src/volume.cpp src/structures.cpp src/fill.cpp src/tin.cpp src/calibration_transform.cpp)
target_link_libraries(pointcloud_compare PRIVATE Qt5::Widgets Qt5::Gui Qt5::Concurrent Qhull::qhullcpp Eigen3::Eigen mesh_volume)
target_compile_options(pointcloud_compare PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(cloud_test src/cloud_test.cpp src/cloud.cpp src/surface_filter.cpp src/columns.cpp src/ply_export.cpp src/volume.cpp src/structures.cpp src/fill.cpp src/tin.cpp src/calibration_transform.cpp)
    target_link_libraries(cloud_test PRIVATE Qt5::Core Qhull::qhullcpp Eigen3::Eigen mesh_volume)
    add_test(NAME raw_measurements COMMAND cloud_test)
    add_executable(fill_test src/fill_test.cpp src/cloud.cpp src/surface_filter.cpp src/columns.cpp src/ply_export.cpp src/volume.cpp src/structures.cpp src/fill.cpp src/tin.cpp src/calibration_transform.cpp)
    target_link_libraries(fill_test PRIVATE Qt5::Core Qhull::qhullcpp Eigen3::Eigen mesh_volume)
    add_test(NAME stockpile_holes COMMAND fill_test)
    add_executable(tin_test src/tin_test.cpp src/tin.cpp src/fill.cpp)
    target_link_libraries(tin_test PRIVATE Qt5::Core Qhull::qhullcpp Eigen3::Eigen mesh_volume)
    add_test(NAME layered_tin COMMAND tin_test)
    add_executable(calibration_test src/calibration_test.cpp src/calibration_transform.cpp src/cloud.cpp src/surface_filter.cpp src/columns.cpp src/ply_export.cpp src/volume.cpp src/structures.cpp src/fill.cpp src/tin.cpp)
    target_link_libraries(calibration_test PRIVATE Qt5::Core Eigen3::Eigen Qhull::qhullcpp mesh_volume)
    add_test(NAME shared_building_frame COMMAND calibration_test $<TARGET_FILE:calibrate_clouds>)
endif()
